C Source Code (tictactoe.c)

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

char board[3][3];
int xWins = 0, oWins = 0, draws = 0;

void resetBoard()
{
    int i, j;
    for (i = 0; i < 3; i++)
        for (j = 0; j < 3; j++)
            board[i][j] = ' ';
}

int checkWin(char p)
{
    int i;
    for (i = 0; i < 3; i++)
    {
        if (board[i][0] == p && board[i][1] == p && board[i][2] == p)
            return 1;
        if (board[0][i] == p && board[1][i] == p && board[2][i] == p)
            return 1;
    }
    if (board[0][0] == p && board[1][1] == p && board[2][2] == p)
        return 1;
    if (board[0][2] == p && board[1][1] == p && board[2][0] == p)
        return 1;
    return 0;
}

int isFull()
{
    int i, j;
    for (i = 0; i < 3; i++)
        for (j = 0; j < 3; j++)
            if (board[i][j] == ' ')
                return 0;
    return 1;
}

void makeMove(char p)
{
    int pos, row, col, c;
    while (1)
    {
        printf("Player %c, choose (1-9): ", p);
        if (scanf("%d", &pos) != 1)
        {
            while ((c = getchar()) != '\n' && c != EOF);
            printf("Invalid input!\n");
            continue;
        }
        if (pos < 1 || pos > 9)
        {
            printf("Must be 1-9.\n");
            continue;
        }
        row = (pos - 1) / 3;
        col = (pos - 1) % 3;
        if (board[row][col] != ' ')
        {
            printf("Position taken!\n");
            continue;
        }
        board[row][col] = p;
        break;
    }
}

/* main(): random start, loop until win/draw,
   then ask "Play again?" (see tictactoe.c) */
